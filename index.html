// Sample product data let products = [ { id: 1, name: "Product 1", price: 29.99, image: "https://via.placeholder.com/150" }, { id: 2, name: "Product 2", price: 49.99, image: "https://via.placeholder.com/150" }, { id: 3, name: "Product 3", price: 19.99, image: "https://via.placeholder.com/150" }, { id: 4, name: "Product 4", price: 99.99, image: "https://via.placeholder.com/150" } ]; let cart = []; let adminAuthenticated = false; // Load products on page load window.onload = () => { displayProducts(); document.getElementById('admin-login-btn').onclick = showAdminLogin; document.getElementById('close-admin').onclick = closeAdminLogin; document.getElementById('admin-login-submit').onclick = adminLogin; document.getElementById('add-product-btn').onclick = showAddProduct; document.getElementById('close-add').onclick = closeAddProduct; document.getElementById('save-product').onclick = saveProduct; document.getElementById('logout-admin').onclick = logoutAdmin; document.getElementById('checkout-btn').onclick = checkout; }; // Display products function displayProducts() { const container = document.getElementById('products-container'); container.innerHTML = ''; products.forEach(product => { const card = document.createElement('div'); card.className = 'product-card'; card.innerHTML = `

${product.name}

$${product.price.toFixed(2)}

`; container.appendChild(card); }); } // Add to cart function addToCart(productId) { const product = products.find(p => p.id === productId); if (product) { cart.push(product); updateCart(); } } // Update cart display function updateCart() { document.getElementById('cart-count').innerText = cart.length; const cartItemsDiv = document.getElementById('cart-items'); cartItemsDiv.innerHTML = ''; if (cart.length === 0) { cartItemsDiv.innerHTML = '

Your cart is empty.

'; return; } let total = 0; cart.forEach((item, index) => { total += item.price; const itemDiv = document.createElement('div'); itemDiv.innerHTML = `

${item.name} - $${item.price.toFixed(2)}

`; cartItemsDiv.appendChild(itemDiv); }); // Show total const totalDiv = document.createElement('div'); totalDiv.innerHTML = `Total: $${total.toFixed(2)}`; cartItemsDiv.appendChild(totalDiv); } // Remove item from cart function removeFromCart(index) { cart.splice(index, 1); updateCart(); } // Checkout function checkout() { if (cart.length === 0) { alert('Your cart is empty.'); return; } alert('Thank you for your purchase!'); cart = []; updateCart(); } // Admin login modal function showAdminLogin() { document.getElementById('admin-modal').style.display = 'flex'; } function closeAdminLogin() { document.getElementById('admin-modal').style.display = 'none'; document.getElementById('admin-message').innerText = ''; document.getElementById('admin-password').value = ''; } // Admin login validation function adminLogin() { const password = document.getElementById('admin-password').value; if (password === 'admin123') { adminAuthenticated = true; document.getElementById('admin-modal').style.display = 'none'; showAdminPanel(); } else { document.getElementById('admin-message').innerText = 'Incorrect password!'; } } // Show admin panel function showAdminPanel() { document.querySelector('.admin-panel').classList.remove('hidden'); loadAdminProducts(); } // Load admin products function loadAdminProducts() { const adminDiv = document.getElementById('admin-products'); adminDiv.innerHTML = ''; products.forEach(product => { const div = document.createElement('div'); div.innerHTML = `

${product.name} - $${product.price.toFixed(2)}

`; adminDiv.appendChild(div); }); } // Logout admin function logoutAdmin() { adminAuthenticated = false; document.querySelector('.admin-panel').classList.add('hidden'); } // Show add product modal function showAddProduct() { document.getElementById('add-product-modal').style.display = 'flex'; } function closeAddProduct() { document.getElementById('add-product-modal').style.display = 'none'; document.getElementById('product-name').value = ''; document.getElementById('product-price').value = ''; } // Save new product function saveProduct() { const name = document.getElementById('product-name').value; const price = parseFloat(document.getElementById('product-price').value); if (name && !isNaN(price)) { const newProduct = { id: Date.now(), name, price, image: "https://via.placeholder.com/150" }; products.push(newProduct); displayProducts(); closeAddProduct(); } else { alert('Please enter valid product details.'); } }
